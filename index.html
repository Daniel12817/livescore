function displayMatches(matches) { const container = document.getElementById('scores'); if (!matches || matches.length === 0) { container.innerHTML = 'No matches'; return; } container.innerHTML = matches.map(m => { const home = m.homeTeam?.name || m.homeTeam || 'Home'; const away = m.awayTeam?.name || m.awayTeam || 'Away'; // Fix the [object Object] issue let scoreText = ''; if (m.score) { if (typeof m.score === 'string') scoreText = m.score; else if (m.score.home != null) scoreText = `${m.score.home} - ${m.score.away}`; else if (m.homeGoals != null) scoreText = `${m.homeGoals} - ${m.awayGoals}`; else scoreText = `${m.goals?.home || 0} - ${m.goals?.away || 0}`; } else { scoreText = m.state || m.status || 'Live'; } return `
${home} vs ${away}
${scoreText} ${m.league?.name || ''} - ${m.status || ''}
`; }).join(''); }
